#include <Arduino.h>

#define RELAY_PIN 25
#define WATER_PIN 26

#define RELAY_ON HIGH
#define RELAY_OFF LOW


#define WATER_OK_LEVEL LOW

void setup()
{
  Serial.begin(115200);

  pinMode(RELAY_PIN, OUTPUT);
  pinMode(WATER_PIN, INPUT_PULLUP);

  digitalWrite(RELAY_PIN, RELAY_OFF);

  Serial.println("Test relay + phun suong + muc nuoc");
}

void loop()
{
  bool hasWater = digitalRead(WATER_PIN) == WATER_OK_LEVEL;

  Serial.print("Water = ");
  Serial.print(hasWater ? "CO NUOC" : "THIEU NUOC");

  if (hasWater)
  {
    digitalWrite(RELAY_PIN, RELAY_ON);
    Serial.println(" | Relay ON | Mist ON");
  }
  else
  {
    digitalWrite(RELAY_PIN, RELAY_OFF);
    Serial.println(" | Relay OFF | Mist OFF");
  }

  delay(1000);
}#include <Arduino.h>
